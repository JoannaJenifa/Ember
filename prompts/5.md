### 3. Create lib/ember/queries.ts (Seller Queries)

```tsx
export interface Seller {
  address: string;
  shopName: string;
  description: string;
  category: number;
  status: number;
  totalSales: number;
  ratingSum: number;
  ratingCount: number;
  youtubeChannel: string;
  createdAt: number;
}

export async function getSeller(sellerAddress: string): Promise<Seller | null> {
  const result = await aptos.view({
    payload: {
      function: `${EMBER_ADDRESS}::seller_registry::get_seller`,
      functionArguments: [sellerAddress],
    },
  });
  return parseSellerResponse(result[0]);
}
```
