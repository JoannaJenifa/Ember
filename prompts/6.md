### 2. Create app/products/components/product-card.tsx

```tsx
import { Product } from '@/lib/ember/product-queries';
import { Card, CardContent } from '@/components/ui/card';
import { Badge } from '@/components/ui/badge';
import { Star } from 'lucide-react';
import Link from 'next/link';

export function ProductCard({ product }: { product: Product }) {
  const avgRating = product.ratingCount > 0
    ? (product.ratingSum / product.ratingCount).toFixed(1)
    : 'New';

  return (
    <Link href={`/product/${product.id}`}>
      <Card className="overflow-hidden hover:shadow-lg transition-shadow">
        <div className="aspect-square relative">
          <img src={product.imageUrl || '/placeholder.png'} alt={product.title} className="object-cover w-full h-full" />
          {product.inventory === 0 && (
            <Badge className="absolute top-2 right-2 bg-destructive">Sold Out</Badge>
          )}
        </div>
        <CardContent className="p-4">
          <h3 className="font-semibold truncate">{product.title}</h3>
          <p className="text-lg font-bold text-primary">{(product.price / 1e8).toFixed(2)} MOVE</p>
          <div className="flex items-center gap-1 text-sm text-muted-foreground">
            <Star className="h-4 w-4 fill-yellow-400 text-yellow-400" />
            <span>{avgRating}</span>
            <span>({product.ratingCount})</span>
          </div>
        </CardContent>
      </Card>
    </Link>
  );
}
```
